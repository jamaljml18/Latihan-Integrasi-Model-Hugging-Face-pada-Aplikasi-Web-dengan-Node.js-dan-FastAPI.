        type="text"
        id="tags"
        placeholder="Tag (pisahkan dengan koma)"
        required
      /><br />
      <textarea id="body" placeholder="Isi catatan" required></textarea><br />
      <button type="submit">Tambah Catatan</button>
    </form>

    <h2>Daftar Catatan</h2>
    <div id="notes-container"></div>

    <script>
      async function loadNotes() {
        const res = await fetch('/notes');
        const data = await res.json();

        const container = document.getElementById('notes-container');
        container.innerHTML = '';

        data.data.notes.forEach((note) => {
          const el = document.createElement('div');
          el.className = 'note';
          el.innerHTML = `
          <h3>${note.title}</h3>
          <p><strong>Tag:</strong> ${note.tags.join(', ')}</p>
          <p>${note.body}</p>
        `;
          container.appendChild(el);
        });
      }

      document
        .getElementById('note-form')
        .addEventListener('submit', async (e) => {
          e.preventDefault();

          const title = document.getElementById('title').value;
          const tags = document
            .getElementById('tags')
            .value.split(',')
            .map((t) => t.trim());
          const body = document.getElementById('body').value;

          await fetch('/notes', {
            method: 'POST',
            headers: { 'Content-Type': 'application/json' },
            body: JSON.stringify({ title, tags, body }),
          });

          // Bersihkan form dan reload
          e.target.reset();
          loadNotes();
        });

      loadNotes();
    </script>
  </body>
